function setupSheetHeaders(sheet) {
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Reference Code", "Timestamp", "Full Name", "Email", "Phone / WhatsApp",
      "Organization", "Industry Sector", "Primary Service Area", "Format / Mode",
      "Preferred Date", "Time Window", "Scope Description", "Source"
    ]);
    var headerRange = sheet.getRange(1, 1, 1, 13);
    headerRange.setFontWeight("bold");
    headerRange.setBackground("#001f3f");
    headerRange.setFontColor("#ffffff");
    sheet.setFrozenRows(1);
  }
}

function processBooking(data) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  setupSheetHeaders(sheet);

  var ref = data.referenceCode || ("MWA-BK-" + Utilities.formatDate(new Date(), "GMT+3", "yyyyMM-") + Math.floor(1000 + Math.random() * 9000));
  
  sheet.appendRow([
    ref,
    data.timestamp || Utilities.formatDate(new Date(), "GMT+3", "yyyy-MM-dd'T'HH:mm:ss'Z'"),
    data.fullName || data.name || "",
    data.email || "",
    data.phone || "",
    data.organization || "Not specified",
    data.sector || "General",
    data.serviceArea || data.service || "",
    data.consultationMode || data.mode || "In-Person (Vision Plaza, Nairobi)",
    data.preferredDate || data.date || "Flexible",
    data.preferredTimeWindow || data.time || "Business Hours",
    data.message || "",
    data.source || "MWA Website"
  ]);

  return { status: "success", referenceCode: ref, message: "Consultation appended to Google Sheet" };
}

function doPost(e) {
  try {
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try { data = JSON.parse(e.postData.contents); } catch (err) { data = e.parameter || {}; }
    } else if (e && e.parameter) {
      data = e.parameter;
    }
    var res = processBooking(data);
    return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    if (e && e.parameter && (e.parameter.name || e.parameter.email || e.parameter.referenceCode)) {
      var res = processBooking(e.parameter);
      return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "ready", firm: "Mwangi Wangondu & Associates" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function syncAllExistingConsultations() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  setupSheetHeaders(sheet);

  var url = "https://raw.githubusercontent.com/Mwangi-Wangondu-Associates/Mwangi-Wangondu-Associates/main/data/consultations.json";
  var res = UrlFetchApp.fetch(url);
  var bookings = JSON.parse(res.getContentText());

  var existingRefs = {};
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    var refs = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < refs.length; i++) {
      if (refs[i][0]) existingRefs[refs[i][0]] = true;
    }
  }

  var importedCount = 0;
  for (var i = bookings.length - 1; i >= 0; i--) {
    var b = bookings[i];
    if (!existingRefs[b.referenceCode]) {
      sheet.appendRow([
        b.referenceCode,
        b.receivedAt || "",
        b.fullName || b.name || "",
        b.email || "",
        b.phone || "",
        b.organization || "Not specified",
        b.sector || "General",
        b.serviceArea || b.service || "",
        b.consultationMode || "In-Person",
        b.preferredDate || b.date || "Flexible",
        b.preferredTimeWindow || b.time || "Business Hours",
        b.message || "",
        "MWA Historical Ledger"
      ]);
      importedCount++;
    }
  }

  SpreadsheetApp.getUi().alert("Import Complete: " + importedCount + " consultations loaded into sheet!");
}

Method 2: Live Auto-Sync to Excel

To have new bookings automatically flow into Excel without downloading files every time, use one of these two proven paths:

Path A: Direct Public CSV Live Feed (Instant & Most Reliable)

Query this permanent public raw feed directly in Microsoft Excel — no Google login or Apps Script permissions required:

  1. In Microsoft Excel, click Data > From Web.
  2. Paste the URL above and click OK > Load.
  3. Under Queries & Connections > Properties, check "Refresh every 5 minutes".
  4. Result: All 16 consultations (including Daniel Minjire, Dr. Peter Kariuki, Esther Mwangi) load immediately into Excel!
Path B: Connected Google Sheet → Excel Sync

If your Excel query showed "0 rows loaded" from pub?output=csv, your Google Sheet has not yet received rows because Apps Script requires public access:

  1. Open your Google Sheet at Your Connected Sheet.
  2. Click Extensions > Apps Script.
  3. Under function dropdown, select syncAllExistingConsultations and click Run. (This immediately fills all 16 rows into your Google Sheet).
  4. Click Deploy > Manage Deployments > Edit (pencil) > set Who has access: "Anyone" > Deploy.
  5. Return to Microsoft Excel and click Refresh All!
